មិនថាអ្នកជាអ្នកអភិវឌ្ឍន៍ខាងក្រោយ, វិស្វករជំនាញ AI, ឬស្ថាបត្យករទិន្នន័យ, មគ្គុទេសក៍នេះផ្តល់ជូននូវការរៀនដែលមានរចនាសម្ព័ន្ធជាមួយគំរូពិតប្រាកដ និងហ្វឹកហាត់ដោយដៃ ដែលនាំអ្នកតាមម៉ាស៊ីនមេ MCP ខាងក្រោម https://github.com/microsoft/MCP-Server-and-PostgreSQL-Sample-Retail។

ជ្រើសរើសផ្លូវរបស់អ្នកដោយផ្អែកលើកម្រិតបទពិសោធន៍របស់អ្នក៖

#### 🟢 **ផ្លូវអ្នកចាប់ផ្តើម** (អ្នកថ្មី MCP)
1. ប្រាកដថាអ្នកបានបញ្ចប់ 0-10 នៃ [MCP សម្រាប់អ្នកចាប់ផ្តើម](https://aka.ms/mcp-for-beginners) ជាមុន
2. បញ្ចប់មន្ទីរ 00-03 ដើម្បីកែលម្អការយល់ដឹងមូលដ្ឋានរបស់អ្នក
3. អនុវត្តមន្ទីរ 04-06 ដើម្បីសាងសង់ដោយដៃ
4. ព្យាយាមមន្ទីរ 07-09 សម្រាប់ករណីប្រើប្រាស់ជាក់ស្តែង

#### 🟡 **ផ្លូវមធ្យម** (មានបទពិសោធ MCP ខ្លះ)
1. ពិនិត្យមន្ទីរ 00-01 សម្រាប់គំនិតពិសេសមូលដ្ឋានទិន្នន័យ
2. ផ្តោតលើមន្ទីរ 02-06 សម្រាប់ការអនុវត្ត
3. ស្ទាត់ជ្រាបជាលึกក្នុងមន្ទីរ 07-12 សម្រាប់លក្ខណៈពិសេសកម្រិតខ្ពស់

#### 🔴 **ផ្លូវចុងខ្ពស់** (មានបទពិសោធ MCP)
1. អានមន្ទីរ 00-03 សម្រាប់បរិបទ
2. ផ្តោតលើមន្ទីរ 04-09 សម្រាប់ការបញ្ចូលមូលដ្ឋានទិន្នន័យ
3. ផ្តោតលើមន្ទីរ 10-12 សម្រាប់ការចេញផ្សាយផលិតកម្ម

## 🛠️ របៀបប្រើផ្លូវរៀននេះយ៉ាងមានប្រសិទ្ធភាព

### រៀនដោយលំដាប់ (បានណែនាំ)

ធ្វើការដំណើរការមន្ទីរតាមលំដាប់សម្រាប់ការយល់ដឹងពេញលេញ៖

1. **អានទិដ្ឋភាពទូទៅ** - យល់ពីអ្វីដែលអ្នកនឹងរៀន
2. **ពិនិត្យលក្ខខណ្ឌមុន** - ធានាថាអ្នកមានចំណេះដឹងត្រូវការ
3. **អនុវត្តតាមមគ្គុទេសក៍ជំហាន** - ប្រតិបត្តិនៅពេលអ្នករៀន
4. **បញ្ចប់ហ្វឹកហាត់** - រឹតបន្តឹងការយល់ដឹងរបស់អ្នក
5. **ពិនិត្យការយកចិត្តទុកដាក់សំខាន់ៗ** - បង្កើតសមត្ថភាពការរៀន

### រៀនជាបំណង

ប្រសិនបើអ្នកត្រូវការជំនាញជាក់លាក់៖

- **ការបញ្ចូលមូលដ្ឋានទិន្នន័យ**: ផ្តោតលើមន្ទីរ 04-06
- **ការអនុវត្តសុវត្ថិភាព**: ផ្តោតលើមន្ទីរ 02, 08, 12
- **AI/ស្វែងរកមានអត្ថន័យ**: ច្រៀងជ្រៅក្នុងមន្ទីរ 07
- **ចេញផ្សាយផលិតកម្ម**: សិក្សាមន្ទីរ 10-12

### អនុវត្តដោយដៃ

មន្ទីរ​រាល់មួយរួមមាន៖
- **គំរូកូដកំពុងដំណើរការ** - ចម្លង ប្តូរ និងសាកល្បង
- **គំរូពិភាក្សាពិតប្រាកដ** - ករណីប្រើប្រាស់វិភាគលក់ជាក់ស្តែង
- **ស្មុគស្មាញដោយជំហាន** - សាងសង់ពីសាមញ្ញទៅកម្រិតខ្ពស់
- **ជំហានការបញ្ចាក់** - បញ្ជាក់ការអនុវត្តរបស់អ្នកដំណើរការ
